cmake_minimum_required(VERSION 3.15)

# Nama project dan bahasa yang digunakan
project(AplikasiKasir LANGUAGES CXX)

# Gunakan standar C++17
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Beri tahu CMake lokasi folder file header (.h)
include_directories(include)

# Kumpulkan semua file .cpp di dalam folder src
file(GLOB SOURCES "src/*.cpp")

# Buat file executable bernama kasir.exe dari sumber di atas
add_executable(kasir ${SOURCES})
